/** * BioGlow Run Planner -> Google Sheet + email * 1) Adds one row per saved match to the score sheet, matching its columns: * Timestamp | Team Name | Total Score | Inspection Bonus | M01 ... M15 | Precision Tokens | Average Time in Launch * 2) Checks the team's earlier totals in the score sheet and marks a new high score * (gold, bold Total Score cell with a note showing the previous best). * Also remembers each team's last average launch time (Script Properties) for the image's ↑/↓ comparison. * 3) Emails the match image (JPEG) to the team's address(es) from the email sheet. * 4) Saves a copy of every match image to a Google Drive folder. * Runs as its own Apps Script project, so it never interferes with other scripts or forms on these sheets. */ const SPREADSHEET_ID = "1utrkYE3YpWn-AKgY5lhoFmQ7FGvEQHhXBwFSDBGJVdA"; // score sheet const SHEET_NAME = "Sheet1"; const EMAIL_SHEET_ID = "1Xnxe3nCVWoz-JDj2odCe6xJnOntK8K1nFNJs1JmlBQk"; // team email sheet const EMAIL_TAB = ""; // tab name, or "" for the first tab const SEND_EMAIL = true; // set to false to stop emailing without changing the app const SAVE_TO_DRIVE = true; // set to false to stop saving images to Drive const DRIVE_FOLDER_ID = ""; // paste a Drive folder's ID here, or leave "" to use a folder // named "BioGlow Table Scoring Images" that the script creates in your My Drive function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const d = JSON.parse(e.postData.contents); const pts = n => (Number(n) || 0) + " pts"; const row = [new Date(), String(d.team || ""), Number(d.total) || 0, pts(d.inspection)]; for (let id = 1; id <= 15; id++) row.push(pts(d.missions && d.missions[id])); row.push(pts(d.precision)); const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SHEET_NAME); const before = teamBest(sheet, String(d.team || "")); // checked before adding this match const total = Number(d.total) || 0; const high = before.count > 0 && total > before.best; sheet.appendRow(row); const r = sheet.getLastRow(); sheet.getRange(r, 1).setNumberFormat("M/d/yyyy HH:mm:ss"); // "Average Time in Launch" column (found by its header), in seconds, e.g. 12.5 shows as "12.5 sec". // Left blank when the robot never returned to a launch area and left again. const avgCol = findColumn(sheet, "Average Time in Launch"); if (avgCol && d.avgLaunch != null) sheet.getRange(r, avgCol).setValue(Number(d.avgLaunch)).setNumberFormat('0.0" sec"'); if (high) sheet.getRange(r, 3).setBackground("#ffd966").setFontWeight("bold") .setNote("New high score! Previous best: " + before.best); d.highScore = high; d.previousBest = before.best; // remember this match's average launch time so the next match can compare against it if (d.avgLaunch != null) PropertiesService.getScriptProperties().setProperty("avg:" + norm(d.team), String(d.avgLaunch)); // Email is separate: if it fails, the score row is still saved. let email = { sent: 0, note: "" }, drive = { url: "", note: "" }; const blob = d.image ? Utilities.newBlob(Utilities.base64Decode(d.image), "image/jpeg", d.imageName || "BioGlow-results.jpg") : null; // Drive and email are separate: if either fails, the score row is still saved. if (SAVE_TO_DRIVE && blob) { try { drive.url = imageFolder().createFile(blob.copyBlob().setName(blob.getName())).getUrl(); } catch (err) { drive.note = "Saving to Google Drive failed: " + err; } } if (!SEND_EMAIL) email.note = "Email is turned off (SEND_EMAIL = false in the script)."; else if (!blob) email.note = "No image arrived with this score, so no email was sent. Upload the newest index.html to GitHub."; else { try { email = emailTeam(d, blob); } catch (err) { email = { sent: 0, note: "Email failed: " + err }; } } return json({ ok: true, emailed: email.sent, emailNote: email.note, savedToDrive: !!drive.url, driveNote: drive.note, highScore: high, previousBest: before.best, earlierScores: before.count }); } catch (err) { return json({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } } // Column number (1-based) whose row-1 header matches the name, ignoring case and spaces; 0 if missing. function findColumn(sheet, name) { const head = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0]; const i = head.findIndex(h => norm(h) === norm(name)); return i + 1; } // Team names match ignoring case, spaces and punctuation; "Error 404: Chicken Not Found" also matches "Error 404". const norm = s => String(s).toLowerCase().replace(/[^a-z0-9]/g, ""); function sameTeam(cell, team) { const n = norm(cell); return !!n && (n === norm(team) || n === norm(String(team).split(":")[0])); } // Best Total Score (column C) among this team's rows (team name in column B) in the score sheet. function teamBest(sheet, team) { const rows = sheet.getDataRange().getDisplayValues(); let best = 0, count = 0; for (let i = 1; i < rows.length; i++) { if (!sameTeam(rows[i][1], team)) continue; const n = parseFloat(String(rows[i][2]).replace(/[^0-9.\-]/g, "")); if (isNaN(n)) continue; count++; if (n > best) best = n; } return { best, count }; } // Finds the team's row in the email sheet and collects every email address in that row. function findEmails(team) { const ss = SpreadsheetApp.openById(EMAIL_SHEET_ID); const sh = EMAIL_TAB ? ss.getSheetByName(EMAIL_TAB) : ss.getSheets()[0]; const rows = sh.getDataRange().getDisplayValues(); for (const r of rows) { if (!r.some(c => sameTeam(c, team))) continue; const found = []; r.forEach(c => (String(c).match(/[^\s,;<>]+@[^\s,;<>]+\.[^\s,;<>]+/g) || []).forEach(a => found.push(a))); return [...new Set(found)]; } return []; } // The Drive folder for images: DRIVE_FOLDER_ID if set, otherwise one the script creates once and remembers. function imageFolder() { if (DRIVE_FOLDER_ID) return DriveApp.getFolderById(DRIVE_FOLDER_ID); const props = PropertiesService.getScriptProperties(), id = props.getProperty("folderId"); if (id) { try { const f = DriveApp.getFolderById(id); if (!f.isTrashed()) return f; } catch (e) {} } const folder = DriveApp.createFolder("BioGlow Table Scoring Images"); props.setProperty("folderId", folder.getId()); return folder; } function emailTeam(d, blob) { const to = findEmails(String(d.team || "")); if (!to.length) return { sent: 0, note: "No email address found for " + d.team + " in the email sheet." }; MailApp.sendEmail({ to: to.join(","), subject: (d.highScore ? "🏆 New high score! " : "") + d.team + " – BioGlow match results (" + d.total + " points)", htmlBody: "

Hi " + d.team + ",

" + (d.highScore ? "

🏆 New high score! You beat your previous best of " + d.previousBest + " points.

" : "") + "

Here are your results from today's BioGlow practice match: " + d.total + " points. The attached image shows your route map, match timeline and score breakdown.

Great work!

", attachments: [blob] }); return { sent: to.length, note: "" }; } // ?best=Team Name returns that team's best recorded total (used by the app for the high-score badge). function doGet(e) { const team = e && e.parameter && e.parameter.best; if (team) { const b = teamBest(SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(SHEET_NAME), team); const last = PropertiesService.getScriptProperties().getProperty("avg:" + norm(team)); return json({ ok: true, best: b.best, count: b.count, lastAvg: last === null ? null : Number(last) }); } return json({ ok: true, message: "BioGlow sheet connector v5 (scores + email + high scores + Drive + launch time) is running. Emails left today: " + MailApp.getRemainingDailyQuota() }); } function json(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
